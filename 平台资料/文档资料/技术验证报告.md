### 优化方案：工具调用(Tool Use)

```python
# 定义工具
tools = [
    {
        "name": "create_file",
        "description": "创建或修改代码文件",
        "input_schema": {
            "type": "object",
            "properties": {
                "file_path": {"type": "string"},
                "content": {"type": "string"},
                "language": {"type": "string"}
            }
        }
    },
    {
        "name": "run_command",
        "description": "在容器中执行命令",
        "input_schema": {
            "type": "object",
            "properties": {
                "command": {"type": "string"}
            }
        }
    }
]

response = client.messages.create(
    model="claude-sonnet-4-5-20250929",
    max_tokens=4096,
    tools=tools,
    messages=[{
        "role": "user",
        "content": "根据需求开发功能..."
    }]
)

# 处理工具调用
for block in response.content:
    if block.type == "tool_use":
        if block.name == "create_file":
            # 在容器中创建文件
            create_file_in_container(
                container,
                block.input["file_path"],
                block.input["content"]
            )
```

### 步骤2：验证代码解析

创建解析工具：`apps/ai_development/utils/code_parser.py`

```python
import re

def parse_code_blocks(markdown_text):
    """
    从markdown文本中解析代码块

    Returns:
        list: [{'file_path': str, 'language': str, 'content': str}]
    """
    # 匹配: ### 文件: path/to/file.ext
    #       ```language
    #       code content
    #       ```

    pattern = r'###\s*文件:\s*([^\n]+)\n\s*```(\w+)\n(.*?)```'
    matches = re.findall(pattern, markdown_text, re.DOTALL)

    files = []
    for file_path, language, content in matches:
        files.append({
            'file_path': file_path.strip(),
            'language': language.strip(),
            'content': content.strip()
        })

    return files

# 测试
test_md = """
### 文件: src/utils/math.js
```javascript
export function add(a, b) {
    return a + b;
}
```

### 文件: tests/math.test.js
```javascript
import { add } from '../src/utils/math.js';

test('add function', () => {
    expect(add(1, 2)).toBe(3);
});
```
"""

files = parse_code_blocks(test_md)
print(f"✅ 解析到 {len(files)} 个文件")
for file in files:
    print(f"  - {file['file_path']} ({file['language']})")
```

### 1. 修改 `claude_code_controller.py`

改为 `anthropic_api_controller.py`:

```python
import anthropic
import re

class AnthropicAPIController:
    """使用Anthropic API进行AI开发"""

    def __init__(self, container, model, api_key):
        self.container = container
        self.model = model
        self.client = anthropic.Anthropic(api_key=api_key)
        self.conversation_logs = []

    def execute_development(self, requirement_text, requirement_id, requirement_name, on_progress=None):
        """执行开发任务"""
        try:
            if on_progress:
                on_progress(20, 'AI正在分析需求...')

            prompt = self._generate_development_prompt(
                requirement_text, requirement_id, requirement_name
            )

            # 调用Anthropic API
            response = self.client.messages.create(
                model=self.model,
                max_tokens=8000,
                messages=[{
                    "role": "user",
                    "content": prompt
                }]
            )

            content = response.content[0].text
            self.log_output(content)

            if on_progress:
                on_progress(50, 'AI正在生成代码...')

            # 解析代码块
            files = self._parse_code_blocks(content)

            if on_progress:
                on_progress(60, f'AI生成了{len(files)}个文件，正在写入容器...')

            # 在容器中创建文件
            for file_info in files:
                self._create_file_in_container(
                    file_info['file_path'],
                    file_info['content']
                )

            return {
                'success': True,
                'conversation_logs': '\n'.join(self.conversation_logs),
                'files_modified': [f['file_path'] for f in files]
            }

        except Exception as e:
            return {
                'success': False,
                'error': str(e),
                'conversation_logs': '\n'.join(self.conversation_logs)
            }
```
