```javascript
  const result = {
    type: 'generic',
    text: '',
    ref: null,
    attributes: {},
    selectors: []
  }

  // 提取元素类型
  const typeMatch = str.match(/^(\w+)/)
  if (typeMatch) result.type = typeMatch[1]

  // 提取文本内容
  const textMatch = str.match(/"([^"]+)"/)
  if (textMatch) result.text = textMatch[1]

  // 提取ref
  const refMatch = str.match(/\[ref=([^\]]+)\]/)
  if (refMatch) result.ref = refMatch[1]

  // 提取属性
  const cursorMatch = str.match(/\[cursor=([^\]]+)\]/)
  if (cursorMatch) result.attributes.cursor = cursorMatch[1]

  const roleMatch = str.match(/\[role=([^\]]+)\]/)
  if (roleMatch) result.attributes.role = roleMatch[1]

  // 生成选择器
  result.selectors = this.generateSelectors(result)

  return result
}
```

#### 1.3 选择器生成

```javascript
generateSelectors(element) {
  const selectors = []

  // 1. data-ref选择器（优先级最高）
  if (element.ref) {
    selectors.push({
      type: 'data-ref',
      value: `[data-ref="${element.ref}"]`,
      priority: 1
    })
  }

  // 2. 文本选择器
  if (element.text) {
    const escapedText = element.text.replace(/"/g, '\\"')
    switch (element.type) {
      case 'button':
        selectors.push({
          type: 'text',
          value: `button:has-text("${escapedText}")`,
          priority: 2
        })
        break
      case 'link':
      case 'a':
        selectors.push({
          type: 'text',
          value: `a:has-text("${escapedText}")`,
          priority: 2
        })
        break
      // ... 其他类型
    }
  }

  // 3. role选择器
  if (element.attributes.role) {
    selectors.push({
      type: 'role',
      value: `[role="${element.attributes.role}"]`,
      priority: 3
    })
  }

  // 4. tag选择器（优先级最低）
  if (element.type !== 'generic') {
    selectors.push({
      type: 'tag',
      value: element.type,
      priority: 4
    })
  }

  return selectors.sort((a, b) => a.priority - b.priority)
}
```

**验证结果**: ✅ 正确
- 按优先级生成多种选择器
- data-ref优先级最高（最稳定）
- tag选择器优先级最低（最不精确）
- 正确排序返回

```javascript
const updateExecutionPath = (node, sourcePortId, targetPortId) => {
  const data = node.getData()
  const config = data?.config || {}
  const snapshotData = config.snapshotData

  if (!snapshotData || !snapshotData.interactiveElements) return

  // 获取源元素和目标元素信息
  const sourceElementId = sourcePortId.replace('element-', '')
  const targetElementId = targetPortId.replace('element-', '')

  const sourceElement = snapshotData.interactiveElements.find(
    el => el.id === sourceElementId
  )
  const targetElement = snapshotData.interactiveElements.find(
    el => el.id === targetElementId
  )

  if (!sourceElement || !targetElement) return

  // 初始化执行路径
  if (!config.executionPath) {
    config.executionPath = []
  }

  // 查找是否已存在相同的路径步骤
  const existingStepIndex = config.executionPath.findIndex(
    step => step.from.elementId === sourceElementId &&
            step.to.elementId === targetElementId
  )

  // 创建步骤对象
  const step = {
    from: {
      elementId: sourceElement.id,
      elementType: sourceElement.type,
      elementText: sourceElement.text
    },
    to: {
      elementId: targetElement.id,
      elementType: targetElement.type,
      elementText: targetElement.text
    },
    action: 'click',
    value: ''
  }

  // 更新或添加步骤
  if (existingStepIndex !== -1) {
    config.executionPath[existingStepIndex] = step
  } else {
    config.executionPath.push(step)
  }

  // 更新节点数据
  node.setData(data)
}
```

**验证结果**: ✅ 正确
- 正确解析端口ID获取元素ID
- 正确查找元素信息
- 避免重复路径步骤
- 记录完整的步骤信息（from, to, action, value）
- 正确更新节点数据

#### 4.1 拓扑排序

```javascript
topologicalSort(nodes, edges) {
  // 构建邻接表和入度表
  const graph = new Map()
  const inDegree = new Map()

  nodes.forEach(node => {
    graph.set(node.id, [])
    inDegree.set(node.id, 0)
  })

  edges.forEach(edge => {
    const source = edge.source?.cell
    const target = edge.target?.cell

    if (source && target && graph.has(source) && graph.has(target)) {
      graph.get(source).push(target)
      inDegree.set(target, inDegree.get(target) + 1)
    }
  })

  // BFS拓扑排序
  const queue = []
  const result = []

  inDegree.forEach((degree, nodeId) => {
    if (degree === 0) {
      queue.push(nodeId)
    }
  })

  while (queue.length > 0) {
    const nodeId = queue.shift()
    result.push(nodeId)

    const neighbors = graph.get(nodeId) || []
    neighbors.forEach(neighbor => {
      inDegree.set(neighbor, inDegree.get(neighbor) - 1)
      if (inDegree.get(neighbor) === 0) {
        queue.push(neighbor)
      }
    })
  }

  // 如果有环，返回原始顺序
  if (result.length !== nodes.length) {
    console.warn('流程图中存在循环')
    return nodes.map(n => n.id)
  }

  return result
}
```

**验证结果**: ✅ 正确
- 正确实现BFS拓扑排序算法
- 正确构建图结构（邻接表+入度表）
- 正确处理循环依赖（返回原始顺序）
- 保证执行顺序正确

```python
                    'modified_at': file_stat.st_mtime
                })

        return Response({
            'count': len(snapshot_files),

        })
```

```python
class PlaywrightSnapshotDetailView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    def get(self, request, filename):
        snapshot_dir = os.path.join(settings.BASE_DIR, 'playwright_snapshot')
        file_path = os.path.join(snapshot_dir, filename)

        # 安全检查：防止路径遍历攻击
        if not os.path.abspath(file_path).startswith(os.path.abspath(snapshot_dir)):
            return Response(
                {'error': '非法的文件路径'},
                status=status.HTTP_400_BAD_REQUEST
            )

        if not os.path.exists(file_path):
            return Response(
                {'error': '快照文件不存在'},
                status=status.HTTP_404_NOT_FOUND
            )

        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()

            file_stat = os.stat(file_path)

            return Response({
                'filename': filename,
                'content': content,
                'size': file_stat.st_size,
                'created_at': file_stat.st_ctime,
                'modified_at': file_stat.st_mtime
            })
        except Exception as e:
            return Response(
                {'error': f'读取文件失败: {str(e)}'},
                status=status.HTTP_500_INTERNAL_SERVER_ERROR
            )
```

```python
class PlaywrightScriptExecuteView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    def post(self, request):
        script_content = request.data.get('script')

        if not script_content:
            return Response(
                {'error': '脚本内容不能为空'},
                status=status.HTTP_400_BAD_REQUEST
            )

        script_id = str(uuid.uuid4())
        script_filename = f'playwright_test_{script_id}.py'

        temp_dir = tempfile.gettempdir()
        script_path = os.path.join(temp_dir, script_filename)

        try:
            # 写入脚本文件
            with open(script_path, 'w', encoding='utf-8') as f:
                f.write(script_content)

            # 执行脚本
            result = subprocess.run(
                ['python', script_path],
                capture_output=True,
                text=True,
                timeout=300  # 5分钟超时
            )

            # 清理临时文件
            if os.path.exists(script_path):
                os.remove(script_path)

            return Response({
                'script_id': script_id,
                'success': result.returncode == 0,
                'stdout': result.stdout,
                'stderr': result.stderr,
                'returncode': result.returncode
            })

        except subprocess.TimeoutExpired:
            if os.path.exists(script_path):
                os.remove(script_path)

            return Response(
                {'error': '脚本执行超时（超过5分钟）'},
                status=status.HTTP_408_REQUEST_TIMEOUT
            )

        except Exception as e:
            if os.path.exists(script_path):
                os.remove(script_path)

            return Response(
                {'error': f'执行脚本失败: {str(e)}'},
                status=status.HTTP_500_INTERNAL_SERVER_ERROR
            )
```
